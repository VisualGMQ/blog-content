{{- /* ========================================================================== Bilibili 短代码（支持 link / facade / player 三种模式） -------------------------------------------------------------------------- 用法： {{< bilibili BV1xx411c7mD >}} {{< bilibili "https://www.bilibili.com/video/BV1xx411c7mD?p=2" >}} {{< bilibili av170001 mode="link" title="标题" cover="封面URL" >}} 参数： id / bvid / bv / 位置参数 : BV 号、av 号、纯数字 aid，或完整 bilibili 链接 mode : link(默认) 跳转卡片 | facade 点击播放 | player 内嵌播放器 title : 卡片/封面上的标题 cover : 封面图 URL（link / facade 用；留空则自动调接口获取） auto : 是否自动获取标题/封面，默认 true（player 模式不需要，故不请求） page : 分 P，默认 1（链接里的 ?p= 会自动识别） danmaku : 是否显示弹幕，默认 false autoplay : 是否自动播放，默认 false high_quality : 是否高清，默认 true caption : 图注 注意：b23.tv 短链无法在构建时解析，请填 BV / av 号或完整链接。 ========================================================================== */ -}} {{- $truthy := slice "true" "1" "yes" "on" -}} {{- $input := .Get "id" | default (.Get "bvid") | default (.Get "bv") | default (.Get "av") | default (.Get "aid") | default (.Get 0) -}} {{- if not $input -}} {{- errorf "bilibili shortcode: 缺少 BV 号 / av 号 / 链接: %s" .Position -}} {{- end -}} {{- $input = trim (string $input) " " -}} {{- $bvid := "" -}} {{- $aid := "" -}} {{- if findRE "^[Bb][Vv][0-9A-Za-z]+$" $input -}} {{- $bvid = $input -}} {{- else if findRE "^[Aa][Vv][0-9]+$" $input -}} {{- $aid = substr $input 2 -}} {{- else if findRE "^[0-9]+$" $input -}} {{- $aid = $input -}} {{- else -}} {{- with (findRE "[Bb][Vv][0-9A-Za-z]+" $input) -}} {{- $bvid = index . 0 -}} {{- end -}} {{- if not $bvid -}} {{- with (findRE "[Aa][Vv][0-9]+" $input) -}} {{- $aid = substr (index . 0) 2 -}} {{- end -}} {{- end -}} {{- end -}} {{- if and (not $bvid) (not $aid) -}} {{- errorf "bilibili shortcode: 无法从 %q 中识别 BV / av 号: %s" $input .Position -}} {{- end -}} {{- $u := urls.Parse $input -}} {{- $page := "" -}} {{- with .Get "page" -}}{{- $page = string . -}}{{- end -}} {{- if not $page -}} {{- with $u.Query.Get "p" -}}{{- $page = . -}}{{- end -}} {{- end -}} {{- $page = $page | default "1" -}} {{- $danmaku := false -}} {{- with .Get "danmaku" -}}{{- $danmaku = in $truthy (lower .) -}}{{- end -}} {{- $autoplay := false -}} {{- with .Get "autoplay" -}}{{- $autoplay = in $truthy (lower .) -}}{{- end -}} {{- $highQuality := true -}} {{- with .Get "high_quality" -}}{{- $highQuality = in $truthy (lower .) -}}{{- end -}} {{- $mode := lower (.Get "mode" | default "link") -}} {{- $title := .Get "title" | default "" -}} {{- $cover := .Get "cover" | default "" -}} {{- $caption := .Get "caption" | default "" -}} {{- /* 未提供封面/标题时，自动调用 bilibili 接口获取（player 模式不需要） */ -}} {{- $auto := true -}} {{- with .Get "auto" -}}{{- $auto = in $truthy (lower .) -}}{{- end -}} {{- if and $auto (ne $mode "player") (or (eq $cover "") (eq $title "")) -}} {{- $api := printf "https://api.bilibili.com/x/web-interface/view?%s" (cond (ne $bvid "") (printf "bvid=%s" $bvid) (printf "aid=%s" $aid)) -}} {{- $headers := dict "User-Agent" "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0 Safari/537.36" "Referer" "https://www.bilibili.com" -}} {{- $meta := dict -}} {{- with try (resources.GetRemote $api (dict "headers" $headers)) -}} {{- if not .Err -}} {{- $meta = .Value.Content | transform.Unmarshal -}} {{- else -}} {{- warnf "bilibili shortcode: 自动获取视频信息失败（%s）：%s" $input .Err -}} {{- end -}} {{- end -}} {{- with $meta.data -}} {{- if eq $cover "" -}} {{- with .pic -}}{{- $cover = replace . "http://" "https://" -}}{{- end -}} {{- end -}} {{- if eq $title "" -}} {{- with .title -}}{{- $title = . -}}{{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $query := slice -}} {{- if $bvid -}} {{- $query = $query | append (printf "bvid=%s" $bvid) -}} {{- else -}} {{- $query = $query | append (printf "aid=%s" $aid) -}} {{- end -}} {{- $query = $query | append (printf "page=%s" $page) -}} {{- $query = $query | append (printf "high_quality=%d" (cond $highQuality 1 0)) -}} {{- $query = $query | append (printf "danmaku=%d" (cond $danmaku 1 0)) -}} {{- $query = $query | append (printf "autoplay=%d" (cond $autoplay 1 0)) -}} {{- $query = $query | append "as_wide=1" -}} {{- $player := printf "//player.bilibili.com/player.html?%s" (delimit $query "&") -}} {{- $base := cond (ne $bvid "") (printf "https://www.bilibili.com/video/%s" $bvid) (printf "https://www.bilibili.com/video/av%s" $aid) -}} {{- $anchor := cond (eq $page "1") $base (printf "%s?p=%s" $base $page) -}} {{- $logoIcon := `` -}} {{- $playIcon := `` -}} {{- $uid := printf "bili-%s" (substr (md5 $input) 0 8) -}}
